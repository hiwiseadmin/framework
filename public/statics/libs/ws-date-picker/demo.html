WsDatePicker 日期选择器

1. 基本示例

最简单的日期选择器用法


// JavaScript
$('#basic-date').wsDatePicker({
    format: 'yyyy-MM-dd',
    onChange: function(date, value) {
        console.log('选择日期:', date, '值:', value);
    }
});

2. 多种类型

支持日期、时间、日期时间、日期范围等类型


// 日期选择
$('#type-date').wsDatePicker({
    type: 'date',
    format: 'yyyy-MM-dd'
});

// 时间选择
$('#type-time').wsDatePicker({
    type: 'time',
    format: 'HH:mm'
});

// 日期时间选择
$('#type-datetime').wsDatePicker({
    type: 'datetime',
    format: 'yyyy-MM-dd HH:mm'
});

// 日期范围选择
$('#type-range').wsDatePicker({
    type: 'date',
    range: true,
    format: 'yyyy-MM-dd',
    rangeSeparator: ' - '
});

3. 配置选项

丰富的配置选项，满足不同需求


// 自定义格式
$('#custom-format').wsDatePicker({
    format: 'yyyy年MM月dd日',
    value: '2024-01-15'
});

// 设置最小/最大日期
$('#min-max').wsDatePicker({
    min: '2024-01-01',
    max: '2024-12-31',
    disabledDate: function(date) {
        // 禁用周末
        return date.getDay() === 0 || date.getDay() === 6;
    }
});

// 静态显示模式
$('#static-mode').wsDatePicker({
    popup: false,
    showQuickSelect: true
});

5. 方法和事件

通过API控制选择器，监听各种事件

事件日志：

// 初始化
var picker = $('#methods-demo').wsDatePicker({
    onChange: function(date, value) {
        logEvent('onChange', value);
    },
    onSelect: function(date, value) {
        logEvent('onSelect', value);
    },
    onOpen: function() {
        logEvent('onOpen');
    },
    onClose: function() {
        logEvent('onClose');
    }
}).data('wsDatePicker');

// 方法调用
$('#btn-show').click(() => picker.show());
$('#btn-hide').click(() => picker.hide());
$('#btn-toggle').click(() => picker.toggle());
$('#btn-set').click(() => picker.setValue('2024-12-25'));
$('#btn-get').click(() => {
    alert('当前值: ' + picker.getValue());
});
$('#btn-clear').click(() => picker.clear());

// 事件日志函数
function logEvent(event, value) {
    const time = new Date().toLocaleTimeString();
    const message = `[${time}] ${event}: ${value || ''}`;
    $('#log-content').prepend(`${message}
`);
}

6. 完整配置说明

参数 类型 默认值 说明
type string 'date' 选择器类型：'date', 'time', 'datetime', 'year', 'month'
format string 'yyyy-MM-dd' 日期格式：yyyy, yy, MM, M, dd, d, HH, H, mm, m, ss, s
value string/Date null 初始值
range boolean false 是否为范围选择
rangeSeparator string ' - ' 范围分隔符
popup boolean true 是否弹窗方式显示
lang string 'zh-CN' 语言：'zh-CN', 'en'
weekStart number 0 周起始日：0(周日) 到 6(周六)
showToday boolean true 是否显示今天按钮
showClear boolean true 是否显示清空按钮
showConfirm boolean true 是否显示确定按钮
showQuickSelect boolean true 是否显示快速选择
disabledDate function null 禁用日期函数，返回true禁用
onChange function null 值变化回调
onSelect function null 选择回调
onOpen function null 打开回调
onClose function null 关闭回调

8. 进阶用法

高级功能和实际应用场景

至

// 动态禁用日期
$('#advanced-disabled').wsDatePicker({
    disabledDate: function(date) {
        // 禁用今天之前的日期
        const today = new Date();
        today.setHours(0, 0, 0, 0);
        return date < today;
    }
});

// 联动选择器
var startPicker = $('#start-date').wsDatePicker({
    format: 'yyyy-MM-dd',
    onChange: function(date) {
        if (date) {
            // 设置结束日期的最小值为开始日期
            endPicker.setOption('min', date);
        }
    }
}).data('wsDatePicker');

var endPicker = $('#end-date').wsDatePicker({
    format: 'yyyy-MM-dd',
    min: new Date() // 默认最小值为今天
}).data('wsDatePicker');

// 带标记的日期
$('#marked-dates').wsDatePicker({
    mark: {
        '2024-01-01': '元旦',
        '2024-02-10': '春节',
        '2024-05-01': '劳动节',
        '2024-10-01': '国庆节'
    },
    onChange: function(date, value) {
        const marks = {
            '2024-01-01': '元旦',
            '2024-02-10': '春节',
            '2024-05-01': '劳动节',
            '2024-10-01': '国庆节'
        };
        
        if (marks[value]) {
            alert('今天是：' + marks[value]);
        }
    }
});

// 动态更新配置
function updatePickerOptions() {
    const picker = $('#marked-dates').data('wsDatePicker');
    picker.setOption('lang', 'en');
}

9. 移动端适配

在移动设备上的优化体验

在移动设备上：
  • 选择器会居中显示
  • 触摸操作优化
  • 更大的点击区域
  • 滑动选择时间

// 移动端优化配置
$('#mobile-optimized').wsDatePicker({
    type: 'datetime',
    format: 'yyyy-MM-dd HH:mm',
    
    // 移动端特定配置
    mobile: true,
    touchEnabled: true,
    
    // 简化界面
    showFooter: true,
    showHeader: true,
    
    onChange: function(date, value) {
        if (/Mobi|Android|iPhone/i.test(navigator.userAgent)) {
            // 移动端特定处理
            console.log('移动端选择:', value);
        }
    }
});

// 检测移动设备
function isMobileDevice() {
    return /Mobi|Android|iPhone/i.test(navigator.userAgent);
}

// 根据设备类型调整配置
if (isMobileDevice()) {
    $.wsDatePicker.setDefaults({
        popup: false,
        showQuickSelect: false
    });
}

10. 常见问题

通过 value 参数设置默认值：


$('#picker').wsDatePicker({
    value: '2024-01-01',
    format: 'yyyy-MM-dd'
});

使用 disabledDate 回调函数：


$('#picker').wsDatePicker({
    disabledDate: function(date) {
        // 禁用周末
        const day = date.getDay();
        return day === 0 || day === 6;
        
        // 禁用特定日期
        const disabledDates = ['2024-01-01', '2024-05-01'];
        const dateStr = utils.formatDate(date, 'yyyy-MM-dd');
        return disabledDates.includes(dateStr);
    }
});

获取选择器实例后调用 setOption 方法：


var picker = $('#picker').data('wsDatePicker');
picker.setOption('min', '2024-01-01');
picker.setOption('max', '2024-12-31');
picker.setOption('lang', 'en');

与其他表单验证库集成：


// 配合 Bootstrap 5 表单验证
$('#myForm').submit(function(e) {
    e.preventDefault();
    
    const dateValue = $('#date-picker').val();
    if (!dateValue) {
        $('#date-picker').addClass('is-invalid');
        return;
    }
    
    // 验证逻辑
    const date = utils.parseDate(dateValue, 'yyyy-MM-dd');
    if (date < new Date()) {
        alert('不能选择过去的日期');
        return;
    }
    
    this.submit();
});

// 清除验证状态
$('#date-picker').on('change', function() {
    $(this).removeClass('is-invalid');
});

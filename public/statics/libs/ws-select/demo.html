// 初始化选择器
$('#mySelect').wsSelect({
    options: [
        { value: '1', label: '选项1', secondary: '描述1', icon: 'bi bi-app' },
        { value: '2', label: '选项2', secondary: '描述2', icon: 'bi bi-android' }
    ],
    multiple: false,
    searchable: true,
    placeholder: '请选择',
    onChange: function(value, option) {
        console.log('选中值:', value);
    }
});

// 远程数据配置
$('#remoteSelect').wsSelect({
    remoteUrl: '/api/search',
    fieldMapping: {
        value: 'id',
        label: 'name',
        secondary: 'description',
        icon: 'icon_class',
        group: 'category'
    },
    displayMode: 'group',
    multiple: true,
    remoteDelay: 500,
    minSearchLength: 2
});

// 获取值
var value = $('#mySelect').wsSelect('getValue');

// 设置值
$('#mySelect').wsSelect('setValue', '1');

// 更新选项
$('#mySelect').wsSelect('updateOptions', newOptions);

// 启用/禁用
$('#mySelect').wsSelect('enable');
$('#mySelect').wsSelect('disable');

// 设置显示模式
$('#mySelect').wsSelect('setDisplayMode', 'group');

// 设置选择模式
$('#mySelect').wsSelect('setMultiple', true);

// 重新加载远程数据
$('#mySelect').wsSelect('reloadRemoteData', '搜索词');

// 初始化带分页的选择器
$('#mySelect').wsSelect({
    options: [...],
    pagination: true,
    pageSize: 10,
    pageSizeOptions: [10, 20, 50],
    showTotal: true,
    showSizeChanger: true
});

// 初始化树形选择器
$('#treeSelect').wsSelect({
    treeMode: true,
    treeExpandAll: false,
    lastLevelLayout: 'horizontal', // 'vertical' or 'horizontal'
    treeCheckable: true,
    multiple: true
});

// 分页控制
$('#mySelect').wsSelect('goToPage', 2);
$('#mySelect').wsSelect('reloadRemoteData');

// 树形控制
$('#treeSelect').wsSelect('expandAll');
$('#treeSelect').wsSelect('collapseAll');
$('#treeSelect').wsSelect('setLastLevelLayout', 'horizontal');

// 动态切换模式
$('#mySelect').wsSelect('setTreeMode', true);
$('#mySelect').wsSelect('setMultiple', true);

代码示例

                    

                    

                        

// HTML 使用示例
<div data-role="wsselect"
     data-name="user_role"
     data-multiple="true"
     data-options='[{"value":"admin","label":"管理员"},{"value":"editor","label":"编辑"}]'
     data-placeholder="选择角色">
</div>

// JavaScript 编程示例
// 获取组件实例
var wsSelect = $('#mySelect').data('wsSelect');

// 获取隐藏input的值
var hiddenValue = wsSelect.getHiddenInputValue();

// 获取隐藏input的DOM元素
var hiddenInput = wsSelect.getHiddenInput();

// 设置值（自动更新隐藏input）
$('#mySelect').wsSelect('setValue', ['1', '2', '3']);

// 同步到原始select元素
wsSelect.syncToOriginalSelect();

// 表单提交时自动包含隐藏input的值
$('form').on('submit', function(e) {
    e.preventDefault();
    var formData = $(this).serialize();
    console.log('表单数据:', formData);
});

// 自动转换原生select
<select name="category" data-convert-to-wsSelect>
    <option value="">请选择</option>
    <option value="1">分类1</option>
</select>

Bootstrap Layer 组件功能演示

基于 Bootstrap 5、jQuery 和 Bootstrap Icons 的现代化弹层组件

提示： 此页面演示了 Bootstrap Layer 组件的所有功能，点击下方按钮体验不同功能。

基础弹层

// 警告框
wsLayer.alert('这是一个警告消息', {
    icon: 0,
    title: '警告'
});

// 确认框
wsLayer.confirm('确定要执行此操作吗？', {
    icon: 4
}, function(index) {
    wsLayer.msg('您点击了确定');
    wsLayer.close(index);
});

// 消息提示
wsLayer.msg('操作成功！');

// 加载层
const loadIndex = wsLayer.load(1, {
    loadingIconSize: 'fs-3',
    loadingTextColor: 'text-white',
    content: '数据加载中...',
    time: 2000
});

// Toast 通知
wsLayer.toast({
    content: '这是一个 Toast 通知',
    type: 'success',
    time: 5000
});

输入和表单

// 文本输入框
wsLayer.prompt({
    title: '请输入内容',
    formType: 'text',
    placeholder: '请输入文本'
}, function(value, index) {
    if(value) {
        wsLayer.msg('您输入了: ' + value);
    }
    wsLayer.close(index);
});

// 文本域输入框
wsLayer.prompt({
    title: '请输入内容',
    formType: 'textarea',
    placeholder: '请输入多行文本',
    area: ['500px', 'auto']
}, function(value, index) {
    if(value) {
        wsLayer.msg('您输入了: ' + value);
    }
    wsLayer.close(index);
});

// 密码输入框
wsLayer.prompt({
    title: '请输入密码',
    formType: 'password',
    placeholder: '请输入密码'
}, function(value, index) {
    if(value) {
        wsLayer.msg('您输入的密码长度: ' + value.length);
    }
    wsLayer.close(index);
});

高级弹层

// 标签页弹层
wsLayer.tab({
    title: '标签页演示',
    area: ['600px', '400px'],
    tab: [
        {
            title: '标签1',
            content: '<h4>标签1内容</h4><p>这是第一个标签的内容</p>'
        },
        {
            title: '标签2',
            content: '<h4>标签2内容</h4><p>这是第二个标签的内容</p>'
        }
    ]
});

// Iframe 弹层
wsLayer.open({
    type: 2,
    title: 'Iframe 演示',
    content: ['https://example.com', 'yes'],
    area: ['800px', '500px']
});

// 自定义弹层
wsLayer.open({
    type: 1,
    title: '自定义弹层',
    content: '<div class="p-3"><h4>自定义内容</h4><p>这里可以放置任何HTML内容</p></div>',
    area: ['500px', 'auto'],
    btn: ['确定', '取消'],
    yes: function(index) {
        wsLayer.msg('点击了确定');
        wsLayer.close(index);
    }
});

// 多个弹层
for(let i = 1; i <= 3; i++) {
    wsLayer.open({
        title: '弹层 ' + i,
        content: '这是第 ' + i + ' 个弹层',
        offset: [100 + i * 50, 100 + i * 50]
    });
}

提示层

将鼠标悬停或点击下方元素查看提示层：

点击或悬停我显示提示 
// 提示层
wsLayer.tips('这是一个提示内容', '#tipsTarget', {
    tips: 1, // 1:上, 2:右, 3:下, 4:左
    time: 3000
});

// 或者使用 msg 作为提示
wsLayer.msg('这是一个消息提示', {
    time: 2000
});

交互功能

// 可拖拽弹层
wsLayer.open({
    title: '可拖拽弹层',
    content: '尝试拖拽标题栏移动弹层',
    move: true
});

// 可调整大小
wsLayer.open({
    title: '可调整大小',
    content: '尝试拖拽右下角调整大小',
    resize: true
});

// 最小化/最大化
wsLayer.open({
    title: '可最小化/最大化',
    content: '点击右上角按钮进行最小化/最大化',
    minBtn: true,
    maxBtn: true
});

// 点击遮罩关闭
wsLayer.open({
    title: '点击遮罩关闭',
    content: '点击遮罩区域可以关闭弹层',
    shadeClose: true
});

// 异步按钮
wsLayer.open({
    title: '异步操作',
    content: '点击按钮执行异步操作',
    btn: ['异步操作'],
    btnAsync: true,
    yes: function(index, layero) {
        return new Promise(function(resolve) {
            setTimeout(function() {
                wsLayer.msg('异步操作完成');
                resolve();
            }, 2000);
        });
    }
});

管理和控制

// 关闭所有弹层
wsLayer.closeAll();

// 关闭指定类型弹层
wsLayer.closeAll('dialog');

// 关闭最近弹层
wsLayer.closeLast();

// 动态修改标题
const index = wsLayer.open({
    title: '原始标题',
    content: '这个弹层的标题将被修改'
});
setTimeout(function() {
    wsLayer.title('新标题', index);
}, 1000);

// 获取所有实例
const instances = wsLayer.getInstances();
console.log('当前弹层实例:', instances);

配置和国际化

// 全局配置
wsLayer.config({
    anim: true,
    isOutAnim: true,
    move: true,
    shade: true,
    shadeClose: false,
    minBtn: true,
    maxBtn: true
});

// 国际化
wsLayer.setLang('zh-CN');
// 添加自定义语言
wsLayer.addMessages('en-US', {
    confirm: 'OK',
    cancel: 'Cancel',
    defaultTitle: 'Information'
});

// 自定义皮肤
wsLayer.open({
    title: '自定义皮肤',
    content: '这个弹层使用了自定义皮肤',
    skin: 'custom-layer-skin'
});
